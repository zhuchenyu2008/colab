cmake_minimum_required(VERSION 3.22.1)
project(vfs_detector LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ncnn_DIR "${CMAKE_CURRENT_SOURCE_DIR}/ncnn-20261020-android-vulkan/${ANDROID_ABI}/lib/cmake/ncnn")
find_package(ncnn REQUIRED)
add_library(vfs_detector SHARED detector.cpp)
find_library(log-lib log)
find_library(jnigraphics-lib jnigraphics)
target_link_libraries(vfs_detector PRIVATE ncnn ${log-lib} ${jnigraphics-lib} android)
